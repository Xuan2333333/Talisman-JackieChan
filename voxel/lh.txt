public VoxelShape makeShape(){

	shape = Shapes.join(shape, Shapes.box(0.4375, 0, 0.5625, 0.5625, 0.0625, 0.59375), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(0.4375, 0.0625, 0.5625, 0.5625, 0.171875, 0.59375), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(0.40625, 0.0625, 0.4375, 0.4375, 0.171875, 0.5625), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(0.40625, 0, 0.4375, 0.4375, 0.0625, 0.5625), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(0.5625, 0.0625, 0.4375, 0.59375, 0.171875, 0.5625), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(0.5625, 0, 0.4375, 0.59375, 0.0625, 0.5625), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(0.4375, 0, 0.4375, 0.5625, 0.03125, 0.5625), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(0.4375, 0, 0.40625, 0.5625, 0.0625, 0.4375), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(0.4375, 0.0625, 0.40625, 0.5625, 0.171875, 0.4375), BooleanOp.OR);

	return shape;
}
